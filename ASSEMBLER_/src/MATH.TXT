# MATH SUBROUTINES
# ADD-16
# SLL-8
# SLR-8
# MUL-8

@ADD-16:
    # O <- A + B
    # O->OUTPUT, A->INPUT, B-INPUT
    # O,A,B ARE 2-BYTE NUMBERS
	
	# IN/OUT VARIABLES
	%ADD-16_IN-A-HB: 0
	%ADD-16_IN-A-LB: 0
	%ADD-16_IN-B-HB: 0
	%ADD-16_IN-B-LB: 0
	%ADD-16_OUT-O-HB: 0
	%ADD-16_OUT-O-LB: 0

	# CLEAR CARRY FLAG
	CCF
	
	# ADD BYTES-0 AND STORE
	LAA %ADD-16_IN-A-LB
	LAB %ADD-16_IN-B-LB
	ADD
	SVA %ADD-16_OUT-O-LB

	# ADD BYTES-1 (W/CARRY) AND STORE
	LAA %ADD-16_IN-A-HB
	LAB %ADD-16_IN-B-HB
	ADC
	SVA %ADD-16_OUT-O-HB
	
	RET

@SLL-8:
	# O <- A << B
	# O->OUTPUT, A->INPUT, B->INPUT
	# 0,A,B ARE 1-BYTE NUMBERS
	
	# IN/OUT VARIABLES
	%SLL-8_IN-A: 0
	%SLL-8_IN-B: 0
	%SLL-8_OUT-O: 0
	
	# O <- A
	# A-REG <- A-VAR
	LAA %SLL-8_IN-A
	# O-VAR <- A-REG
	SVA %SLL-8_OUT-O
	
	# SHIFT (B-VAR)-TIMES A-REG
	%SLL-8_IT: 0
	@SLL-8_FOR-CYCLE:
		# FOR-CYCLE: CHECK CURRENT CYCLE CONDITION
		LAA %SLL-8_IN-B
		TWO
		TBA
		LAA %SLL-8_IT
		CZF # CLEAR ZERO FLAG
		ADD
		JZZ @SLL-8_FOR-CYCLE-EXIT
		
		# FOR-CYCLE CONTENT HERE
		# A-REG <- O-VAR
		LAA %SLL-8_OUT-O
		SLL
		SVA %SLL-8_OUT-O
		
		# INCREMENT ITERATOR
		LAA %SLL-8_IT
		INA
		SVA %SLL-8_IT
		
		# JUMP TO NEXT CYCLE
		JMP @SLL-8_FOR-CYCLE
	
	@SLL-8_FOR-CYCLE-EXIT:
		RET

@SLR-8:
	# O <- A >> B
	# O->OUTPUT, A->INPUT, B->INPUT
	# 0,A,B ARE 1-BYTE NUMBERS
	
	# IN/OUT VARIABLES
	%SLR-8_IN-A: 0
	%SLR-8_IN-B: 0
	%SLR-8_OUT-O: 0
	
	# O <- A
	# A-REG <- A-VAR
	LAA %SLR-8_IN-A
	# O-VAR <- A-REG
	SVA %SLR-8_OUT-O
	
	# SHIFT (B-VAR)-TIMES A-REG
	%SLR-8_IT: 0
	@SLR-8_FOR-CYCLE:
		# FOR-CYCLE: CHECK CURRENT CYCLE CONDITION
		LAA %SLR-8_IN-B
		TWO
		TBA
		LAA %SLR-8_IT
		CZF # CLEAR ZERO FLAG
		ADD
		JZZ @SLR-8_FOR-CYCLE-EXIT
		
		# FOR-CYCLE CONTENT HERE
		# A-REG <- O-VAR
		LAA %SLR-8_OUT-O
		SLR
		SVA %SLR-8_OUT-O
		
		# INCREMENT ITERATOR
		LAA %SLR-8_IT
		INA
		SVA %SLR-8_IT
		
		# JUMP TO NEXT CYCLE
		JMP @SLR-8_FOR-CYCLE
	
	@SLR-8_FOR-CYCLE-EXIT:
		RET

@MUL-8:
	# O <- A * B
	# O->OUTPUT, A->INPUT, B->INPUT
	# 0,A,B ARE 1-BYTE NUMBERS
	
	# IN/OUT VARIABLES
	%MUL-8_IN-A: 15
	%MUL-8_IN-B: 8
	%MUL-8_OUT-O: 0
	
	# O-VAR CONTAIN THE ACCUMULATE OF THE SUMS
	# O-VAR <- 0
	LVA 0
	SVA %MUL-8_OUT-O
	
	%MUL-8_IT: 0
	@MUL-8_FOR-CYCLE:
		# FOR-CYCLE: CHECK CURRENT CYCLE CONDITION
		LAA %MUL-8_IT
		LVB -8
		CZF # CLEAR ZERO FLAG
		ADD
		JZZ @MUL-8_FOR-CYCLE-EXIT
		
		# FOR-CYCLE CONTENT HERE
		# B-VAR >> IT
		# SET INPUT VARS OF 'SLR-8' SUBROUTINE
		LAA %MUL-8_IN-B
		SVA %SLR-8_IN-A
		LAA %MUL-8_IT
		SVA %SLR-8_IN-B
		SBR @SLR-8
		LAA %SLR-8_OUT-O
		# A-VAR & 0x1
		ANV 0X01
		# DEPENDING OF A-REG VALUE:
		# 0: CONTINUE;
		# 1: %MUL-8_OUT-O += A-REG
		JZZ @MUL-8_IF
			# CASE 1
			LAA %MUL-8_IN-A
			LAB %MUL-8_OUT-O
			ADD
			SVA %MUL-8_OUT-O
			JMP @MUL-8_IF-END
		@MUL-8_IF:
			# CASE 0
			JMP @MUL-8_FOR-CYCLE-NEXT
		@MUL-8_IF-END:
		
		
	@MUL-8_FOR-CYCLE-NEXT:
		# INCREMENT ITERATOR
		LAA %MUL-8_IT
		INA
  		SVA %MUL-8_IT
		
		# JUMP TO NEXT CYCLE
		JMP @MUL-8_FOR-CYCLE
	
	@MUL-8_FOR-CYCLE-EXIT:
		RET
