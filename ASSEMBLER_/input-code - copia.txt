# PROGRAMA QUE RECORRE TODOS LOS ELEMENTOS
# DE UNA LISTA PARA PROBAR SI LAS MENCIONES
# HIGH/LOW DE DATA_LABELS FUNCIONAN O NO

JMP @MAIN

@ADD-16:
    # O <- A + B
    # O->OUTPUT, A->INPUT, B-INPUT
    # O,A,B ARE TWO-BYTE NUMBERS
	
	#IN/OUT VARIABLES
	%ADD-16_IN-A-HB: 0
	%ADD-16_IN-A-LB: 0
	%ADD-16_IN-B-HB: 0
	%ADD-16_IN-B-LB: 0
	%ADD-16_OUT-O-HB: 0
	%ADD-16_OUT-O-LB: 0

	# CLEAR CARRY FLAG
	CCF

	# ADD LOW BYTES AND STORE
	LAA %ADD-16_IN-A-LB
	LAB %ADD-16_IN-B-LB
	ADD
	SVA %ADD-16_OUT-O-LB

	# ADD HIGH BYTES (W/CARRY) AND STORE
	LAA %ADD-16_IN-A-HB
	LAB %ADD-16_IN-B-HB
	ADC
	SVA %ADD-16_OUT-O-HB
	RET

@MAIN:
	# RECORRER LA LISTA 'LIST' Y EN CADA
	# ITERACION GUARDAR SU BYTE EN LA
	# VARIABLE 'VAR' EN RAM
	%LIST: 1 3 5 7 9
	%VAR: 0X0F
	
	
	%IT: 0 # ITERATOR
	@CYCLE:
		# VAR = *(LIST + IT)
		# LLAMAR @ADD-16 CON
		# A <- &LIST (320 -> 0x140)
		# B <- IT (CON SU BYTE MAS ALTO EN CERO)

		# TRANSFER &LIST TO A VAR
		LVA >LIST
		LVB <LIST
		SVA >ADD-16_IN-A-HB <ADD-16_IN-A-HB
		SVB >ADD-16_IN-A-LB <ADD-16_IN-A-LB

		# TRANSFER IT TO B VAR
		LVA 0
		LAB %IT
		SVA >ADD-16_IN-B-HB <ADD-16_IN-B-HB
		SVB >ADD-16_IN-B-LB <ADD-16_IN-B-LB

		# COMPUTE THE ADDITION (r=0x141)
		SBR @ADD-16

		# EL RESULTADO DE LA SUMA ES LA DIRECCION
		# DEL ELEMENTO ACTUAL DEL CICLO FOR QUE SE
		# GUARDARA EN LA VARIABLE 'VAR' EN RAM
		
		# LEER EL BYTE ACTUAL EN LA LISTA
		# Y GUARDARLO EN EL REGISTRO A
		LVA 88
		LAB >ADD-16_OUT-O-HB <ADD-16_OUT-O-HB
		LAC >ADD-16_OUT-O-LB <ADD-16_OUT-O-LB
		#LPA
		SPA

		# GUARDAR REGISTRO A EN RAM EN VARIABLE 'VAR'
		#SVA %VAR
		
		# CHECK FOR CYCLE CONDITION
		LAA %IT
		INA
		SVA %IT
		CZF # CLEAR ZERO FLAG
		ADV -5
		JNZ @CYCLE
		
		
	LVC 0XFF
